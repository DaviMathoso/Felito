-- ============================================================
-- FELITO - CATÁLOGO DE SORVETES
-- Banco: MySQL 8+ / InnoDB / utf8mb4
-- ============================================================

CREATE DATABASE IF NOT EXISTS felito
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE felito;

SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;

-- ============================================================
-- 1) USERS
-- ============================================================
CREATE TABLE IF NOT EXISTS users (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    name VARCHAR(120) NOT NULL,
    email VARCHAR(190) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('customer','admin') NOT NULL DEFAULT 'customer',
    loyalty_points INT NOT NULL DEFAULT 0,
    anonymized_at DATETIME NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_users_email (email),
    CONSTRAINT chk_users_loyalty_points CHECK (loyalty_points >= 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================================
-- 2) FLAVORS
-- ============================================================
CREATE TABLE IF NOT EXISTS flavors (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    name VARCHAR(120) NOT NULL,
    description TEXT NOT NULL,
    price DECIMAL(8,2) NOT NULL,
    stock_qty INT NOT NULL DEFAULT 0,
    min_stock INT NOT NULL DEFAULT 5,
    is_featured TINYINT(1) NOT NULL DEFAULT 0,
    is_active TINYINT(1) NOT NULL DEFAULT 1,
    image_path VARCHAR(255) NULL,
    tags VARCHAR(255) NULL,
    PRIMARY KEY (id),
    CONSTRAINT chk_flavors_price CHECK (price >= 0),
    CONSTRAINT chk_flavors_stock_qty CHECK (stock_qty >= 0),
    CONSTRAINT chk_flavors_min_stock CHECK (min_stock >= 0),
    CONSTRAINT chk_flavors_is_featured CHECK (is_featured IN (0,1)),
    CONSTRAINT chk_flavors_is_active CHECK (is_active IN (0,1))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================================
-- 3) COUPONS
-- ============================================================
CREATE TABLE IF NOT EXISTS coupons (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    code VARCHAR(40) NOT NULL,
    type ENUM('percent','fixed') NOT NULL,
    value DECIMAL(8,2) NOT NULL,
    uses INT NOT NULL DEFAULT 0,
    max_uses INT NULL,
    valid_until DATETIME NULL,
    is_active TINYINT(1) NOT NULL DEFAULT 1,
    PRIMARY KEY (id),
    UNIQUE KEY uq_coupons_code (code),
    CONSTRAINT chk_coupons_value CHECK (value >= 0),
    CONSTRAINT chk_coupons_uses CHECK (uses >= 0),
    CONSTRAINT chk_coupons_max_uses CHECK (max_uses IS NULL OR max_uses >= 0),
    CONSTRAINT chk_coupons_is_active CHECK (is_active IN (0,1))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================================
-- 4) ORDERS
-- ============================================================
CREATE TABLE IF NOT EXISTS orders (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    user_id INT UNSIGNED NOT NULL,
    status ENUM('reserved','paid','expired','cancelled') NOT NULL DEFAULT 'reserved',
    subtotal DECIMAL(10,2) NOT NULL,
    discount DECIMAL(10,2) NOT NULL DEFAULT 0,
    total DECIMAL(10,2) NOT NULL,
    coupon_id INT UNSIGNED NULL,
    reserved_until DATETIME NOT NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    paid_at DATETIME NULL,
    PRIMARY KEY (id),
    KEY idx_orders_user_id (user_id),
    KEY idx_orders_coupon_id (coupon_id),
    KEY idx_orders_status (status),
    KEY idx_orders_reserved_until (reserved_until),
    CONSTRAINT fk_orders_user
        FOREIGN KEY (user_id) REFERENCES users(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
